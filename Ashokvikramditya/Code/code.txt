<script>
  /* ---------- utilities ---------- */
  function toast(msg, isErr){
    const box = document.getElementById('toastBox');
    const t = document.createElement('div');
    t.className = 'toast' + (isErr ? ' err' : '');
    t.innerHTML = '<span class="dot"></span><span>'+msg+'</span>';
    box.appendChild(t);
    setTimeout(()=>{ t.style.opacity='0'; t.style.transform='translateX(30px)'; setTimeout(()=>t.remove(),300); }, 3600);
  }
  function openModal(id){ document.getElementById(id).classList.add('active'); }
  function closeModal(id){ document.getElementById(id).classList.remove('active'); }
  document.querySelectorAll('.overlay').forEach(ov=>{
    ov.addEventListener('click', e=>{ if(e.target===ov) ov.classList.remove('active'); });
  });

  /* ---------- slot picker ---------- */
  function pickSlot(el){
    el.parentElement.querySelectorAll('.slot').forEach(s=>s.classList.remove('selected'));
    el.classList.add('selected');
  }

  /* ---------- auth ---------- */
  function openAuth(tab){
    const user = JSON.parse(localStorage.getItem('healthrapoo_user') || 'null');
    if(user){ openProfile(); return; }
    switchAuthTab(tab);
    document.getElementById('authForms').style.display='block';
    document.getElementById('profileView').style.display='none';
    openModal('authOverlay');
  }
  function switchAuthTab(tab){
    document.getElementById('tabSignin').classList.toggle('active', tab==='signin');
    document.getElementById('tabSignup').classList.toggle('active', tab==='signup');
    document.getElementById('signinForm').classList.toggle('active', tab==='signin');
    document.getElementById('signupForm').classList.toggle('active', tab==='signup');
  }
  function handleSignin(e){
    e.preventDefault();
    const email = e.target.querySelector('input[type=email]').value;
    const name = email.split('@')[0].replace(/[^a-zA-Z]/g,' ').trim() || 'Patient';
    setUser({ name: capitalize(name), email });
    closeModal('authOverlay');
    toast('Signed in — welcome back, '+capitalize(name)+'.');
    return false;
  }
  function handleSignup(e){
    e.preventDefault();
    const inputs = e.target.querySelectorAll('input');
    const first = inputs[0].value, last = inputs[1].value, email = inputs[2].value;
    setUser({ name: first+' '+last, email });
    closeModal('authOverlay');
    toast('Account created — welcome to Healthrapoo, '+first+'.');
    return false;
  }
  function setUser(user){
    const existing = JSON.parse(localStorage.getItem('healthrapoo_user') || 'null');
    user.bookings = existing ? (existing.bookings || 0) : 0;
    localStorage.setItem('healthrapoo_user', JSON.stringify(user));
    renderUser();
  }
  function handleSignout(){
    localStorage.removeItem('healthrapoo_user');
    renderUser();
    closeModal('authOverlay');
    toast('Signed out. See you soon!');
  }
  function openProfile(){
    const user = JSON.parse(localStorage.getItem('healthrapoo_user') || 'null');
    if(!user){ openAuth('signin'); return; }
    document.getElementById('authForms').style.display='none';
    document.getElementById('profileView').style.display='block';
    document.getElementById('profileName').textContent = user.name;
    document.getElementById('profileEmail').textContent = user.email;
    document.getElementById('profileBookingsCount').textContent = user.bookings || 0;
    openModal('authOverlay');
  }
  function renderUser(){
    const user = JSON.parse(localStorage.getItem('healthrapoo_user') || 'null');
    const chip = document.getElementById('userChip');
    const signInBtn = document.getElementById('signInBtn');
    if(user){
      chip.style.display='flex';
      signInBtn.style.display='none';
      document.getElementById('userChipName').textContent = user.name.split(' ')[0];
      document.getElementById('userChipImg').src = 'https://images.pexels.com/photos/32967602/pexels-photo-32967602.jpeg?auto=compress&cs=tinysrgb&w=100';
    } else {
      chip.style.display='none';
      signInBtn.style.display='inline-flex';
    }
  }
  function capitalize(s){ return s.charAt(0).toUpperCase()+s.slice(1); }

  /* ---------- booking ---------- */
  function openBooking(){ openModal('bookOverlay'); }
  function handleFullBooking(e){
    e.preventDefault();
    const user = JSON.parse(localStorage.getItem('healthrapoo_user') || 'null');
    if(user){ user.bookings = (user.bookings||0) + 1; localStorage.setItem('healthrapoo_user', JSON.stringify(user)); renderUser(); }
    const id = 'HR-' + Math.floor(100000 + Math.random()*900000);
    closeModal('bookOverlay');
    toast('Appointment confirmed — reference '+id+'.');
    e.target.reset();
    return false;
  }
  document.getElementById('quickBookForm').addEventListener('submit', function(e){
    e.preventDefault();
    const id = 'HR-' + Math.floor(100000 + Math.random()*900000);
    toast('Slot held — reference '+id+'. Check your email to confirm.');
  });

  function subscribeNews(){
    const val = document.getElementById('newsletterEmail').value;
    if(!val || !val.includes('@')){ toast('Enter a valid email to subscribe.', true); return; }
    toast('Subscribed! Health tips land in your inbox monthly.');
    document.getElementById('newsletterEmail').value='';
  }

  /* ---------- scroll reveal ---------- */
  const io = new IntersectionObserver((entries)=>{
    entries.forEach(en=>{ if(en.isIntersecting){ en.target.classList.add('in'); io.unobserve(en.target); } });
  }, { threshold: 0.15 });
  document.querySelectorAll('.reveal').forEach(el=>io.observe(el));

  renderUser();
</script>
